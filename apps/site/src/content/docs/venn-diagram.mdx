```tsx

import "@carlos-burelo/klin/styles.css";

<ChartTooltipPortal>
<VennDiagram
  sets={/* … */}
  ariaLabel="Venn Diagram"
/>
</ChartTooltipPortal>
```

## Código de esta demo

<details>
<summary>Ver fuente (`src/demos/venn-diagram.tsx`)</summary>

```tsx
import type { VennSet } from "@carlos-burelo/klin";
import { ChartTooltipPortal, VennDiagram } from "@carlos-burelo/klin";

const sets: [VennSet, VennSet, VennSet] = [
  { label: "Correo", value: 5200 },
  { label: "Mensaje de texto", value: 3800 },
  { label: "Aplicación móvil", value: 2900 },
];

const intersections = [
  { sets: [0, 1], value: 1450 },
  { sets: [0, 2], value: 980 },
  { sets: [1, 2], value: 760 },
  { sets: [0, 1, 2], value: 410 },
];

function format(value: number): string {
  return `${new Intl.NumberFormat("es-MX").format(Math.round(value))} personas`;
}

export default function VennDiagramDemo() {
  return (
    <ChartTooltipPortal>
      <VennDiagram
        sets={sets}
        intersections={intersections}
        ariaLabel="Traslape de canales de notificación usados por la ciudadanía"
        format={format}
      />
    </ChartTooltipPortal>
  );
}
```

## Accesibilidad

- Toda gráfica exige `ariaLabel` descriptivo; el SVG usa `role="img"` con `\<title\>`.
- Cada forma lleva `\<title\>` nativo (tooltip sin JS) y `data-chart-tip` para el portal rico.
- Sin datos se renderiza estado vacío homogéneo (`ChartEmpty`), nunca SVG roto.
- Detalle del contrato en [/guias/accesibilidad](/guias/accesibilidad).
